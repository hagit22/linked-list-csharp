מחלקת StringNode

using System; public class StringNode { private string value; private StringNode next; //----------------------------------- //constructors public StringNode(string value) { this.value = value; this.next = null; } public StringNode(string value, StringNode next) { this.value = value; this.next = next; } //----------------------------------- //getters public string GetValue() { return this.value; } public StringNode GetNext() { return this.next; } //----------------------------------- //setters public void SetValue(string value) { this.value = value; } public void SetNext(StringNode next) { this.next = next; } //----------------------------------- //return true if this.next is not null, else returns false public bool HasNext() { return (this.next != null); } //----------------------------------- //ToString public override string ToString() { return value + " --> " + next; } }
